## 四、关键代码变更

### 4.1 VoteFusion（3 行代码）

```go
// kernel/rank_fusion.go Fuse() 函数中
// 原代码：return top1
// 新代码：对 top-3 候选做加权平均
if len(ranked) >= 3 {
    w1, w2, w3 := ranked[0].Score, ranked[1].Score, ranked[2].Score
    total := w1 + w2 + w3
    if total > 0 {
        ranked[0].Score = (w1*w1 + w2*w2*0.5 + w3*w3*0.3) / total
    }
}
```

### 4.3 BudgetManager（10 行代码）

```go
// kernel/lifecycle.go
func (s *APIServer) BudgetManager() {
    limits := map[MemoryLayer]int{LayerL0: 50, LayerL1: 500, LayerL2: 5000}
    for layer, limit := range limits {
        ms, _ := s.backend.ScanLayer(context.Background(), layer, "", 10000)
        if len(ms) > limit {
            sort.Slice(ms, func(i, j int) bool { return ms[i].Value < ms[j].Value })
            for _, m := range ms[:len(ms)-limit] {
                s.backend.Delete(context.Background(), m.MemoryID)
            }
        }
    }
}
```

### 4.4 CoreMemoryManager（5 行代码）

```go
// kernel/lifecycle.go
func (s *APIServer) CoreMemoryManager() {
    for id, count := range s.access_count {
        if count > 10 {
            if m, _ := s.backend.Get(context.Background(), id, "", ""); m != nil {
                m.IsCore = true
            }
        }
    }
}
```
